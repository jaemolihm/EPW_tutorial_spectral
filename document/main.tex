\section*{Introduction}

\section*{Exercise 1: Zero-point renormalization and temperature-dependent band gap of diamond}

\subsection*{1.1 Theory}
In this example we calculate the phonon-induced band gap renormalization of diamond.
In the Allen--Heine--Cardona (AHC) formalism$^{1}$, the phonon-induced renormalization of electron eigenvalue is determined by the real part of the electron self-energy, which is the sum of the Fan and the Debye--Waller (DW) terms:
\begin{equation}
   \Sigma_{n\mb{k}} = \Sigma^{\rm Fan}_{n\mb{k}} + \Sigma^{\rm DW}_{n\mb{k}}.
\end{equation}


\vspace{0.5cm}
The non-adiabatic Fan term comes from the second-order perturbative correction due to the linear electron-phonon coupling parameter and is given by\footnote{S.\ Ponc\'e \textit{et al}., \href{https://pubs.aip.org/aip/jcp/article-abstract/143/10/102813/898920/Temperature-dependence-of-the-electronic-structure?redirectedFrom=fulltext}{J.\ Chem.\ Phys.\ \textbf{143}, 102813 (2015)}}:
\begin{equation}
   \Sigma^{\rm Fan}_{n\mb{k}} = \!\int\! \frac{d^3q}{\Omega_{\mathrm{BZ}}} \sum_{m\nu} | g_{mn\nu}(\mb{k,q})|^2
   \sum_{\pm} \frac{n_{\mb{q}\nu} + f_{m\mb{k}+\mb{q}}^\pm}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}} \pm \omega_{\mb{q}\nu} + i\eta},
\end{equation}
where $f_{m\mb{k}+\mb{q}}^+$ is the Fermi--Dirac occupation factor of the state with energy $\varepsilon_{m\mb{k}+\mb{q}}$, $f_{m\mb{k}+\mb{q}}^- = 1 - f_{m\mb{k}+\mb{q}}^+$, and $\eta$ a positive infinitesimal value (\texttt{degaussw}).
This expression contains an infinite sum over bands $m$ which cannot be computed using Wannier interpolation of low-energy bands.


To solve this problem, we use Wannier function perturbation theory (WFPT).
We begin by splitting the Fan term into the low-energy active space contribution ($\Sigma^{\rm A-Fan}$) and the rest space contribution ($\Sigma^{\rm R-Fan}$) separated by the cutoff energy $E^{\rm A}$:
\begin{align}
   \label{eq:fan_active}
   \Sigma^{\rm A-Fan}_{n\mb{k}}
   &= \!\int\! \frac{d^3q}{\Omega_{\mathrm{BZ}}} \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} \leq E^{\rm A}}} \sum_{\nu} | g_{mn\nu}(\mb{k,q})|^2
   \sum_{\pm} \frac{n_{\mb{q}\nu} + f_{m\mb{k}+\mb{q}}^\pm}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}} \pm \omega_{\mb{q}\nu} + i\eta},
   \\
   \label{eq:fan_rest}
   \Sigma^{\rm R-Fan}_{n\mb{k}}
   &= \!\int\! \frac{d^3q}{\Omega_{\mathrm{BZ}}} \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} > E^{\rm A}}} \sum_{\nu} | g_{mn\nu}(\mb{k,q})|^2
   \frac{2n_{\mb{q}\nu} + 1}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}} \pm \omega_{\mb{q}\nu} + i\eta}.
\end{align}
In Eq.~\eqref{eq:fan_rest}, we assumed that the cutoff energy is much higher than the Fermi level such that the electron occupation is always zero above the window.
For states $n\mb{k}$ within the active space, $\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}}$ in the denominator of Eq.~\eqref{eq:fan_rest} is never zero.
Assuming this electronic energy difference is much larger than the phonon frequency, we can drop the phonon frequency and approximate the rest term as
\begin{equation} \label{eq:fan_rest_adia}
   \Sigma^{\rm R-Fan}_{n\mb{k}} \approx \!\int\! \frac{d^3q}{\Omega_{\mathrm{BZ}}} \sum_{\nu} (2n_{\mb{q}\nu} + 1) \!\!\!\!
   \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} > E^{\rm A}}} \frac{| g_{mn\nu}(\mb{k,q})|^2}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}}}.
\end{equation}
In practice, $E^{\rm A}$ (\texttt{ahc\_win\_max}) is a convergence parameter that needs to be increased until the result is converged.
Usually, 1~eV above the highest band of interest is sufficient.


The infinite sum over the high-energy bands in Eq.~\eqref{eq:fan_rest_adia} can be rewritten as\footnote{X.\ Gonze \textit{et al}., \href{https://onlinelibrary.wiley.com/doi/10.1002/andp.201000100}{Ann. Phys. 523, 168 (2011)}}
\begin{equation} \label{eq:sternheimer_mel}
   \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} > E^{\rm A}}} \frac{| g_{mn\nu}(\mb{k,q})|^2}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}}}
   = \langle \partial^{\rm R}_{\mb{q}\nu} u_{n\mb{k}} | \partial^{\rm R}_{\mb{q}\nu} u_{n\mb{k}} \rangle,
\end{equation}
where $| \partial^{\rm R}_{\mb{q}\nu} u_{n\mb{k}} \rangle$ is the first-order perturbation of the electron wavefunction projected onto the high-energy subspace, which can be computed by solving the Sternheimer equation
\begin{equation} \label{eq:sternheimer}
   (\varepsilon_{n\mb{k}} - \hat{H}_{\mb{k}+\mb{q}}) | \partial^{\rm R}_{\mb{q}\nu} u_{n\mb{k}} \rangle
   =  \hat{P}^{\rm R}_{\mb{k+q}} \,  \partial_{\mb{q}\nu} \hat{V} | u_{n\mb{k}} \rangle.
\end{equation}
Here, $\hat{P}^{\rm R}_{\mb{k+q}}$ is the projector onto the high-energy bands.


Using WFPT, the wavefunction perturbation $| \partial^{\rm R}_{\mb{q}\nu} u_{n\mb{k}} \rangle$ can be represented in terms of localized objects.
These objects are called ``Wannier function perturbations'', as they are the change of Wannier functions due to a phonon perturbation.
Correspoingly, the matrix element on the right-hand side of \Eq{eq:sternheimer_mel} can be Wannier interpolated from a coarse $\mb{k}$-point grid to real space, and then to a dense $\mb{k}$-point grid.
More information about WFPT can be found in the article \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.11.041053}{J.\ M.\ Lihm and C.\ H.\ Park, Phys. Rev. X \textbf{11}, 041053 (2021)}.



\vspace{0.5cm}
The DW term comes from the first-order perturbative correction due to the quadratic electron-phonon coupling parameter $D_{n\nu}(\mb{k,q})$:
\begin{equation}
   \Sigma^{\rm DW}_{n\mb{k}} = \sum_{m\nu} \!\int\! \frac{d^3q}{\Omega_{\mathrm{BZ}}} \frac{n_{\mb{q}\nu} + 1/2}{2\omega_{\mb{q}\nu}} D_{n\nu}(\mb{k,q}).
\end{equation}
The $D_{n\nu}(\mb{k,q})$ term, which is called the DW matrix element, is defined as:
\begin{equation}
   D_{n\nu}(\mb{k,q}) = \sum_{\kappa\kappa'\alpha\alpha'} e^*_{\kappa\alpha\nu}(\mb{q}) e_{\kappa'\alpha'\nu}(\mb{q})
   \langle u_{n\mb{k}} | \partial_{-\mb{q}\kappa\alpha} \partial_{\mb{q}\kappa'\alpha'} \hat{V} | u_{n\mb{k}} \rangle.
\end{equation}
This term cannot be directly evaluated within first-order DFPT because it involves the second derivative of the electronic potential with respect to atomic displacements.
Using the rigid-ion approximation which neglects the cross-derivatives between different atoms, and exploiting the translational covariance of the electron potential, the DW matrix element can be expressed as\footnote{J.-M.\ Lihm and C.-H.\ Park, \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.101.121102}{Phys.\ Rev.\ B \textbf{101}, 121102(R) (2020)}}
\begin{equation}
   \langle u_{n\mb{k}} | \partial_{-\mb{q}\kappa\alpha} \partial_{\mb{q}\kappa'\alpha'} \hat{V} | u_{n\mb{k}} \rangle
   \stackrel{\rm RIA}{\approx}
   i \delta_{\kappa\kappa'} \langle u_{n\mb{k}} | [\partial_{\mb{0}\kappa\alpha} \hat{V}, \hat{p}_{\alpha'}] | u_{n\mb{k}}\rangle,
\end{equation}
where $\hat{p}_{\alpha'} = -i \partial / \partial r_{\alpha'}$ is the momentum operator in the direction $\alpha'$.
Therefore, in the RIA, the DW self-energy takes the following explicit form:
\begin{equation}
   \Sigma^{\rm DW}_{n\mb{k}}
   \stackrel{\rm RIA}{\approx}
   \sum_{m\nu} \!\int\! \frac{d^3q}{\Omega_{\mathrm{BZ}}} \frac{n_{\mb{q}\nu} + 1/2}{2\omega_{\mb{q}\nu}}
   \sum_{\kappa\alpha\alpha'} e^*_{\kappa\alpha\nu}(\mb{q}) e_{\kappa\alpha'\nu}(\mb{q})
   \langle u_{n\mb{k}} | [\partial_{\mb{0}\kappa\alpha} \hat{V}, \hat{p}_{\alpha'}] | u_{n\mb{k}} \rangle\,.
\end{equation}
Note the $\kappa$, the atomic index, is the same for the two eigenmodes as the RIA neglects inter-atomic cross-derivatives.
% More information can be found in the article \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.101.121102}{Phys. Rev. B \textbf{101}, 121102(R) (2020)}.

\end{footnotesize}

\section*{Exercise 3: Phononic spectral function of MgB$_2$}

In this example, we calculate the phonon spectral function of MgB$_2$.
We mostly highlight the specificity of phonon spectral function calculations and it is therefore advised to first do exercise 2.

The 1-loop (1L) contribution to the phonon self-energy is given by\footnote{M.\ Calandra \textit{et al}., \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.82.165111}{Phys.\ Rev.\ B \textbf{82}, 165111 (2010)}, J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}}
\begin{equation} \label{eq:phself}
   \Pi^{\rm 1L}_\qnu(\omega; T) = \frac{g_{\rm s}}{N_k} \sum_{mn\mb{k}} \frac{f_{n\mb{k}}(T) - f_{m\mb{k+q}}(T)}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k+q}} + \omega + i\eta} \lvert g_{mn\nu}(\mb{k}, \mb{q}) \rvert^2.
\end{equation}
Here, $g_{\rm s}$ is the spin degeneracy factor.
In principle, the self-energy $\Pi^{\rm 1L}_{\mb{q}\mu\nu}(\omega; T)$ also has off-diagonal terms that couple different phonon modes and change the phonon eigenmodes\footnote{J.\ Berges \textit{et al}., \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.101.155107}{Phys. Rev. B \textbf{101}, 155107 (2020)}, see Fig.~2(c).}, but we neglect them here: $\Pi^{\rm 1L}_{\mb{q}\mu\nu} \stackrel{\Delta}{=} \Pi^{\rm 1L}_{\mb{q}\nu} \delta_{\mu \nu}$.
Calculation of the off-diagonal self-energy will be made available in the EPW code in the near future.

\vspace{0.2cm}

Since the DFPT calculation already includes static screening, we have to subtract the static DFPT contribution from the phonon self-energy to avoid double counting.
Also, in this ``unscreening'' process, we have to take into account that the DFPT calculation is done usually at a high temperature (e.g., $T^{\rm high} = 8000$~K) and the phonon self-energy is calculated at a lower temperature (e.g., $T^{\rm low} = 300$~K).
Hence, we subtract the static self-energy at $T^{\rm high}$, and then add the dynamical self-energy at $T^{\rm low}$:
\begin{equation} \label{eq:phself_unscreening}
   \Delta \Pi_\qnu(\omega; T^{\rm low}) = \Pi^{\rm 1L}_\qnu(\omega; T^{\rm low}) - \Pi^{\rm 1L}_\qnu(0; T^{\rm high})
\end{equation}
More information about the phonon self-energy and the unscreening procedure can be found in the article J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}.

\vspace{0.2cm}

The phonon spectral function reads\footnote{P.\ B.\ Allen and R.\ Silberglitt, \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.9.4733}{Phys.\ Rev.\ B \textbf{9}, 4733 (1974)}. See also Eqs.~(2,3) of J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}. We have an additional $2\omega_\qnu$ factor multiplied to the self-energy because of the difference in the definition of the electron-phonon matrix element $g_{mn\nu}(\mb{k}, \mb{q})$. Compare Eq.~(2) of F.\ Giustino, \href{https://journals.aps.org/rmp/abstract/10.1103/RevModPhys.89.015003}{Rev. Mod. Phys. \textbf{89}, 015003 (2017)} and Eq.~(12) of J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}.}
\begin{equation} \label{eq:phspec}
   A_{\mb{q}}(\omega; T) = -\frac{2\omega}{\pi} \sum_{\nu} \mathrm{Im} \frac{1}{(\omega + i\eta)^2 - \omega_\qnu^2 - 2\omega_\qnu \, \Delta \Pi_\qnu(\omega; T)}
\end{equation}

#!/usr/bin/env python3
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.image import NonUniformImage

plt.rcParams.update({'font.size': 15})

def parse_epw_selfen(filename, nk, nbnd):
    xks = np.zeros((3, nk))
    energy = np.zeros((nbnd, nk))
    sigma = np.zeros((nbnd, nk), dtype=complex)

    with open(filename, 'r') as f:
        for line in f:
            if "Electron Self-Energy using Wannier function perturbation theory" in line:
                break
        print(f.readline())
        f.readline()
        f.readline()
        for ik in range(nk):
            xks[:, ik] = [float(x) for x in f.readline().split()[-3:]]
            f.readline()
            for ib in range(nbnd):
                data = f.readline().split()
                energy[ib, ik] = float(data[3])
                sigma[ib, ik] = float(data[6]) + 1j * float(data[9])
            f.readline()
            f.readline()
            f.readline()

    sigma /= 1000 # meV to eV
    return xks, energy, sigma


def parse_epw_specfun_sup(filename):
    data = np.loadtxt(filename)
    nk = int(data[-1, 0])
    nbnd = int(data[-1, 1])

    data = data.reshape(nbnd, nk, -1, 6)
    ws = data[0, 0, :, 3]
    es = data[:, :, 0, 2]
    sigma = (data[:, :, :, 4] + 1j * data[:, :, :, 5]) / 1000 # meV to eV
    return ws, es, sigma

def parse_epw_specfun(filename):
    data = np.loadtxt(filename)
    nk = int(data[-1, 0])

    data = data.reshape(nk, -1, 3)
    ws = data[0, :, 1]
    As = data[:, :, 2] * 1000  # 1/meV to 1/eV

    n_int = []
    with open(filename, "r") as f:
        for line in f:
            if "Integrated spectral function" in line:
                n_int += [float(line.split()[-1])]
    n_int = np.array(n_int)

    return ws, As, n_int

nbnd, nk, nfreq = sigma.shape

xks, _, sigma_ahc = parse_epw_selfen("epw2.out", nk, nbnd)

# A = 1 / (w - e - sigma)

#!/usr/bin/env python3
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.image import NonUniformImage

plt.rcParams.update({'font.size': 15})

filename = "specfun_sup.phon"

# --------------------------------------
# Parse specfun_sup file
data = np.loadtxt(filename)
nq = int(data[-1, 0])
nmodes = int(data[-1, 1])

# [iq, iT, iw, imode, idata]
data = data.reshape(nq, 2, -1, nmodes, 8)
ws = data[0, 0, :, 0, 4] * 1e3  # eV to meV
nw = len(ws)

T_low = data[0, 0, 0, 0, 2]
T_high = data[0, 1, 0, 0, 2]
w_ph = data[:, 0, 0, :, 3] * 1e3  # eV to meV
Pi_low = data[:, 0, :, :, 5] + 1j * data[:, 0, :, :, 7]
Pi_high_0 = data[:, 1, :, :, 6]
Pi = Pi_low - Pi_high_0

xks = np.loadtxt("qpt.txt", skiprows=1)[:, :3].T
dks = np.linalg.norm(xks[:, 1:] - xks[:, :-1], axis=0)
xs = np.concatenate(([0.], np.cumsum(dks)))

ws_plot = np.linspace(ws.min(), ws.max(), 1_000, True)

# --------------------------------------
# Compute self-energy
A = np.zeros((nq, len(ws_plot)))
w_ph_low = np.zeros_like(w_ph)
eta = 0.1 # meV

for iq in range(nq):
   for imode in range(nmodes):
      wq = w_ph[iq, imode]

      w2 = wq**2 + 2 * wq * Pi[iq, 0, imode]
      w_ph_low[iq, imode] = np.sqrt(abs(w2).real) * np.sign(w2.real)

      Pi_itp = np.interp(ws_plot, ws, Pi[iq, :, imode])

      for iw in range(len(ws_plot)):
         w = ws_plot[iw]
         Pi_w = Pi_itp[iw]
         A[iq, iw] += -np.imag(2 * w / ((w + 1j * eta)**2 - wq**2 - 2 * wq * Pi_w)) / np.pi

A[A < 0] = 1e-10

fig, axes = plt.subplots(2, 2, figsize=(10, 6), sharex="col",
   gridspec_kw={'height_ratios': [3, 1], "width_ratios":[1,0.02]})
axes[1, 1].set_axis_off()

# Plot spectral function
plt.sca(axes[0, 0])
im = NonUniformImage(axes[0, 0], interpolation='nearest',
