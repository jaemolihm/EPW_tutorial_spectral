\section*{Introduction}

In this tutorial, we show how to compute the phonon-induced renormalization of electron band structure of diamond within the Allen--Heine--Cardona (AHC) theory\footnote{P.\ B.\ Allen and V.\ Heine, \href{https://iopscience.iop.org/article/10.1088/0022-3719/9/12/013}{J.\ Phys.\ C, \textbf{9}, 2305 (1976)}, P.\ B.\ Allen and M.\ Cardona, \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.23.1495}{Phys.\ Rev.\ B, \textbf{23}, 1495 (1981)}}

the electron spectral function of MgB$_2$ and of graphene (with a comparison to the experimental self-energy extracted via \href{https://xarpes.readthedocs.io/en/latest}{xARPES}), and the phonon spectral function of MgB$_2$.

\section*{Exercise 1: Zero-point renormalization and temperature-dependent band gap of diamond}

\subsection*{1.1 Theory}
In this example we calculate the phonon-induced band gap renormalization of diamond.
In the AHC formalism$^{1}$, the phonon-induced renormalization of electron eigenvalue is determined by the real part of the electron self-energy, which is the sum of the Fan and the Debye--Waller (DW) terms:
\begin{equation}
   \Sigma_{n\mb{k}} = \Sigma^{\rm Fan}_{n\mb{k}} + \Sigma^{\rm DW}_{n\mb{k}}.
\end{equation}


\vspace{0.5cm}
The non-adiabatic Fan term comes from the second-order perturbative correction due to the linear electron-phonon coupling parameter and is given by\footnote{S.\ Ponc\'e \textit{et al}., \href{https://pubs.aip.org/aip/jcp/article-abstract/143/10/102813/898920/Temperature-dependence-of-the-electronic-structure?redirectedFrom=fulltext}{J.\ Chem.\ Phys.\ \textbf{143}, 102813 (2015)}}:
\begin{equation}
   \Sigma^{\rm Fan}_{n\mb{k}} = \!\int\! \frac{\mathrm{d}\mathbf{q}}{\Omega^{\mathrm{BZ}}} \sum_{m\nu} | g_{mn\nu}(\mb{k,q})|^2
   \sum_{\pm} \frac{n_{\nu\mb{q}} + f_{m\mb{k}+\mb{q}}^\pm}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}} \pm \omega_{\nu\mb{q}} + i\eta},
\end{equation}
where $f_{m\mb{k}+\mb{q}}^+$ is the Fermi--Dirac occupation factor of the state with energy $\varepsilon_{m\mb{k}+\mb{q}}$, $f_{m\mb{k}+\mb{q}}^- = 1 - f_{m\mb{k}+\mb{q}}^+$, $n_{\nu\mathbf{q}}$ is the Bose--Einstein occupation, and $\eta$ a positive infinitesimal value (\texttt{degaussw}).
%
This expression contains an infinite sum over bands $m$ which cannot be computed using Wannier interpolation of low-energy bands.


To solve this problem, we use WFPT.
We begin by splitting the Fan term into the low-energy active space contribution ($\Sigma^{\rm A-Fan}$) and the rest space contribution ($\Sigma^{\rm R-Fan}$) separated by the cutoff energy $E^{\rm A}$:
\begin{align}
   \label{eq:fan_active}
   \Sigma^{\rm A-Fan}_{n\mb{k}}
   &= \!\int\! \frac{\mathrm{d}\mathbf{q}}{\Omega^{\mathrm{BZ}}} \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} \leq E^{\rm A}}} \sum_{\nu} | g_{mn\nu}(\mb{k,q})|^2
   \sum_{\pm} \frac{n_{\nu\mb{q}} + f_{m\mb{k}+\mb{q}}^\pm}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}} \pm \omega_{\nu\mb{q}} + i\eta} \,,
   \\
   \label{eq:fan_rest}
   \Sigma^{\rm R-Fan}_{n\mb{k}}
   &= \!\int\! \frac{\mathrm{d}\mathbf{q}}{\Omega^{\mathrm{BZ}}} \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} > E^{\rm A}}} \sum_{\nu} | g_{mn\nu}(\mb{k,q})|^2
   \frac{2n_{\nu\mb{q}} + 1}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}} \pm \omega_{\nu\mb{q}} + i\eta} \,.
\end{align}
%
In Eq.~\eqref{eq:fan_rest}, we assume that the cutoff energy is much higher than the Fermi level such that the electron occupation is always zero above the window.
%
For states $n\mb{k}$ within the rest space, the $(\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}})$ term in the denominator of Eq.~\eqref{eq:fan_rest} is never zero.
%
Assuming this electronic energy difference is much larger than the phonon frequency, we can drop the phonon frequency and approximate the rest term as
\begin{equation} \label{eq:fan_rest_adia}
   \Sigma^{\rm R-Fan}_{n\mb{k}} \approx \!\int\! \frac{\mathrm{d}\mathbf{q}}{\Omega^{\mathrm{BZ}}} \sum_{\nu} (2n_{\nu\mb{q}} + 1) \!\!\!\!
   \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} > E^{\rm A}}} \frac{| g_{mn\nu}(\mb{k,q})|^2}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}}} \,.
\end{equation}
In practice, $E^{\rm A}$ (\texttt{ahc\_win\_max}) is a convergence parameter that needs to be increased until the result is converged.
Usually, 1~eV above the highest band of interest is sufficient.


The infinite sum over the high-energy bands in Eq.~\eqref{eq:fan_rest_adia} can be rewritten as\footnote{X.\ Gonze \textit{et al}., \href{https://onlinelibrary.wiley.com/doi/10.1002/andp.201000100}{Ann. Phys. 523, 168 (2011)}}
\begin{equation} \label{eq:sternheimer_mel}
   \sum_{m}^{\varepsilon_{m\mb{k}+\mb{q} > E^{\rm A}}} \frac{| g_{mn\nu}(\mb{k,q})|^2}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k}+\mb{q}}}
   = \langle \partial^{\rm R}_{\nu\mb{q}} u_{n\mb{k}} | \partial^{\rm R}_{\nu\mb{q}} u_{n\mb{k}} \rangle,
\end{equation}
where $| \partial^{\rm R}_{\nu\mb{q}} u_{n\mb{k}} \rangle$ is the first-order perturbation of the electron wavefunction projected onto the high-energy subspace, which can be computed by solving the Sternheimer equation
\begin{equation} \label{eq:sternheimer}
   (\varepsilon_{n\mb{k}} - \hat{H}_{\mb{k}+\mb{q}}) | \partial^{\rm R}_{\nu\mb{q}} u_{n\mb{k}} \rangle
   =  \hat{P}^{\rm R}_{\mb{k+q}} \,  \partial_{\nu\mb{q}} \hat{V} | u_{n\mb{k}} \rangle.
\end{equation}
Here, $\hat{P}^{\rm R}_{\mb{k+q}}$ is the projector onto the high-energy bands.


Using WFPT, the wavefunction perturbation $| \partial^{\rm R}_{\nu\mb{q}} u_{n\mb{k}} \rangle$ can be represented in terms of localized objects.
These objects are called ``Wannier function perturbations'', as they are the change of Wannier functions due to a phonon perturbation.
Correspondingly, the matrix element on the right-hand side of \Eq{eq:sternheimer_mel} can be Wannier interpolated from a coarse $\mb{k}$-point grid to real space, and then to a dense $\mb{k}$-point grid.
More information about WFPT can be found in the article \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.11.041053}{J.\ M.\ Lihm and C.\ H.\ Park, Phys. Rev. X \textbf{11}, 041053 (2021)}.



\vspace{0.5cm}
The DW term comes from the first-order perturbative correction due to the quadratic electron-phonon coupling parameter $D_{n\nu}(\mb{k,q})$:
\begin{equation}
   \Sigma^{\rm DW}_{n\mb{k}} = \sum_{m\nu} \!\int\! \frac{\mathrm{d}\mathbf{q}}{\Omega^{\mathrm{BZ}}} \frac{n_{\nu\mb{q}} + 1/2}{2\omega_{\nu\mb{q}}} D_{n\nu}(\mb{k,q}).
\end{equation}
The $D_{n\nu}(\mb{k,q})$ term, which is called the DW matrix element, is defined as:
\begin{equation}
   D_{n\nu}(\mb{k,q}) = \sum_{\kappa\kappa'\alpha\alpha'} e^*_{\kappa\alpha\nu}(\mb{q}) e_{\kappa'\alpha'\nu}(\mb{q})
   \langle u_{n\mb{k}} | \partial_{-\mb{q}\kappa\alpha} \partial_{\mb{q}\kappa'\alpha'} \hat{V} | u_{n\mb{k}} \rangle.
\end{equation}
This term cannot be directly evaluated within first-order DFPT because it involves the second derivative of the electronic potential with respect to atomic displacements.
Using the rigid-ion approximation (RIA) which neglects the cross-derivatives between different atoms, and exploiting the translational covariance of the electron potential, the DW matrix element can be expressed as\footnote{J.-M.\ Lihm and C.-H.\ Park, \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.101.121102}{Phys.\ Rev.\ B \textbf{101}, 121102(R) (2020)}}
\begin{equation}
   \langle u_{n\mb{k}} | \partial_{-\mb{q}\kappa\alpha} \partial_{\mb{q}\kappa'\alpha'} \hat{V} | u_{n\mb{k}} \rangle
   \stackrel{\rm RIA}{\approx}
   i \delta_{\kappa\kappa'} \langle u_{n\mb{k}} | [\partial_{\mb{0}\kappa\alpha} \hat{V}, \hat{p}_{\alpha'}] | u_{n\mb{k}}\rangle,
\end{equation}
where $\hat{p}_{\alpha'} = -i \partial / \partial r_{\alpha'}$ is the momentum operator in the direction $\alpha'$.
Therefore, in the RIA, the DW self-energy takes the following explicit form:
\begin{equation}
   \Sigma^{\rm DW}_{n\mb{k}}
   \stackrel{\rm RIA}{\approx}
   \sum_{m\nu} \!\int\! \frac{\mathrm{d}\mathbf{q}}{\Omega^{\mathrm{BZ}}} \frac{n_{\nu\mb{q}} + 1/2}{2\omega_{\nu\mb{q}}}
   \sum_{\kappa\alpha\alpha'} e^*_{\kappa\alpha\nu}(\mb{q}) e_{\kappa\alpha'\nu}(\mb{q})
   \langle u_{n\mb{k}} | [\partial_{\mb{0}\kappa\alpha} \hat{V}, \hat{p}_{\alpha'}] | u_{n\mb{k}} \rangle\,.
\end{equation}
Note that $\kappa$, the atomic index, is the same for the two eigenmodes as the RIA neglects inter-atomic cross-derivatives.
% More information can be found in the article \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.101.121102}{Phys. Rev. B \textbf{101}, 121102(R) (2020)}.

\end{footnotesize}

From the underlined numbers, we find that the renormalization of the valence band maxima (VBM), lowest conduction band at $\mb{k}=\Gamma$, and conduction band minima (CBM) are 124.73, -350.68, and -167.00 meV, respectively.

The computed spectral function can be visualized using the \program{plot.py} script.
Since the spectral function $A_{n\mb{k}}(\omega)$ has sharp peaks at the quasiparticle energy, the \texttt{nw\_specfun = 200} grid written out by EPW is too coarse to resolve them cleanly.
The self-energy $\Sigma_{n\mb{k}}(\omega)$, however, is a smooth function of $\omega$.
The \program{plot.py} script therefore reads $\Sigma_{n\mb{k}}(\omega)$ from \texttt{specfun\_sup.elself.300.000K}, linearly interpolates it onto a much denser frequency grid (10\,000 points), and evaluates the spectral function on that dense grid via
\begin{equation}
   A_{n\mb{k}}(\omega) = -\frac{1}{\pi} \mathrm{Im}\, \frac{1}{\omega - \varepsilon_{n\mb{k}} - \Sigma_{n\mb{k}}(\omega)},
   \qquad A_{\mb{k}}(\omega) = \sum_n A_{n\mb{k}}(\omega).
\end{equation}
This recovers the sharp quasiparticle peaks without re-running EPW on a finer \texttt{nw\_specfun} grid.
To use matplotlib, we should first load the module and install the matplotlib package.

\begin{verbatim}
$ module load python/3.9.2

$ python3 plot.py
\end{verbatim}

\section*{Exercise 3: Electronic spectral function of graphene}

In this example, we compute the electron spectral function of graphene from electron-phonon coupling and compare it with self-energy extracted from the ARPES spectrum done in the \href{https://xarpes.readthedocs.io/en/latest/notebooks/graphene.html}{xARPES tutorial}.
%
The formalism and workflow are identical to Exercise~2, so we focus on the practical differences specific to graphene.
%
Tomorrow, we will re-use graphene in the spectral transport hands-on to compute its optical conductivity.

\vspace{0.2cm}

Graphene is a semimetal with a linear Dirac dispersion at the $K$ point.
%
Coupling to the in-plane $E_{2g}$ optical phonon ($\hbar \omega^{\rm ph} \approx 200$~meV) produces a kink in the renormalized band structure roughly one phonon energy below the Fermi level\footnote{See, e.g., A.\ Bostwick \textit{et al.}, \href{https://www.nature.com/articles/nphys477}{Nature Phys.\ \textbf{3}, 36 (2007)} and references therein.}.


In the experiment of D.~Yu.~Usachov \textit{et al.}\footnote{D.~Yu.~Usachov \textit{et al.}, \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.97.085132}{Phys.\ Rev.\ B \textbf{97}, 085132 (2018)}.}, the authors used Li adsorption to dope electrons into graphene, shifting the Fermi level upward by approximately $1.6$~eV, close to the van Hove singularity at the $M$ point.
%
Due to the large density of states near the van Hove singularity, the electron self-energy is strongly enhanced.
%
In this exercise, we will try to reproduce the experimental self-energy and the kink in the spectral function by rigidly shifting the Fermi level in our calculation.

\section*{Exercise 4: Phononic spectral function of MgB$_2$}

In this example, we calculate the phonon spectral function of MgB$_2$.
We mostly highlight the specifics of phonon spectral function calculations and it is therefore advised to first do Exercise 2.

The 1-loop (1L) contribution to the phonon self-energy is given by\footnote{M.\ Calandra \textit{et al}., \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.82.165111}{Phys.\ Rev.\ B \textbf{82}, 165111 (2010)}, J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}}
\begin{equation} \label{eq:phself}
   \Pi^{\rm 1L}_\nuq(\omega; T) = \frac{g_{\rm s}}{N_k} \sum_{mn\mb{k}} \frac{f_{n\mb{k}}(T) - f_{m\mb{k+q}}(T)}{\varepsilon_{n\mb{k}} - \varepsilon_{m\mb{k+q}} + \omega + i\eta} \lvert g_{mn\nu}(\mb{k}, \mb{q}) \rvert^2.
\end{equation}
Here, $g_{\rm s}$ is the spin degeneracy factor.
In principle, the self-energy $\Pi^{\rm 1L}_{\mb{q}\mu\nu}(\omega; T)$ also has off-diagonal terms that couple different phonon modes and change the phonon eigenmodes\footnote{J.\ Berges \textit{et al}., \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.101.155107}{Phys. Rev. B \textbf{101}, 155107 (2020)}, see Fig.~2(c).}, but we neglect them here: $\Pi^{\rm 1L}_{\mb{q}\mu\nu} \stackrel{\Delta}{=} \Pi^{\rm 1L}_{\nu\mb{q}} \delta_{\mu \nu}$.
Calculation of the off-diagonal self-energy will be made available in the EPW code in the near future.

\vspace{0.2cm}

Since the DFPT calculation already includes static screening, we have to subtract the static DFPT contribution from the phonon self-energy to avoid double counting.
Also, in this ``unscreening'' process, we have to take into account that the DFPT calculation is done usually at a high temperature (e.g., $T^{\rm high} = 8000$~K) and the phonon self-energy is calculated at a lower temperature (e.g., $T^{\rm low} = 300$~K).
Hence, we subtract the static self-energy at $T^{\rm high}$, and then add the dynamical self-energy at $T^{\rm low}$:
\begin{equation} \label{eq:phself_unscreening}
   \Delta \Pi_\nuq(\omega; T^{\rm low}) = \Pi^{\rm 1L}_\nuq(\omega; T^{\rm low}) - \Pi^{\rm 1L}_\nuq(0; T^{\rm high}) \,.
\end{equation}
More information about the phonon self-energy and the unscreening procedure can be found in the article J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}.

\vspace{0.2cm}

The phonon spectral function reads\footnote{P.\ B.\ Allen and R.\ Silberglitt, \href{https://journals.aps.org/prb/abstract/10.1103/PhysRevB.9.4733}{Phys.\ Rev.\ B \textbf{9}, 4733 (1974)}. See also Eqs.~(2,3) of J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}. We have an additional $2\omega_\nuq$ factor multiplied to the self-energy because of the difference in the definition of the electron-phonon matrix element $g_{mn\nu}(\mb{k}, \mb{q})$. Compare Eq.~(2) of F.\ Giustino, \href{https://journals.aps.org/rmp/abstract/10.1103/RevModPhys.89.015003}{Rev. Mod. Phys. \textbf{89}, 015003 (2017)} and Eq.~(12) of J.\ Berges \textit{et al}., \href{https://journals.aps.org/prx/abstract/10.1103/PhysRevX.13.041009}{Phys. Rev. X \textbf{13}, 041009 (2023)}.}
\begin{equation} \label{eq:phspec}
   B_{\mb{q}}(\omega; T) = -\frac{2\omega}{\pi} \sum_{\nu} \mathrm{Im} \frac{1}{(\omega + i\eta)^2 - \omega_\nuq^2 - 2\omega_\nuq \, \Delta \Pi_\nuq(\omega; T)}
\end{equation}

\paragraph{Automatic unscreening (EPW v6.0+).}
Since EPW v6.0 (released August 2025), the phonon self-energy at the high temperature $T^{\rm high}$ used for the unscreening in \Eq{eq:phself_unscreening} is computed \emph{automatically}, with the same smearing method and parameter that were used in the \texttt{scf} calculation.
%
For example, here the \texttt{scf.in} uses Methfessel--Paxton smearing (\texttt{smearing = `mp'}) with \texttt{degauss}=0.05~Ry.
%
The same smearing is reused internally for $\Pi^{\rm 1L}_\nuq(0; T^{\rm high})$.
%
The user therefore only has to specify the low temperatures in the EPW \texttt{temps} input, which corresponds to the physical temperature with a Fermi--Dirac occupation function.

\vspace{0.2cm}

\begin{footnotesize}
   {\bf Notes}:
   \begin{itemize}
      \item We set \texttt{lopt\_w2b = .false.} because the $\mb{q}$-point is not from a homogeneous grid, so the optimization does not work. We can then use the memory-optimized version with \texttt{etf\_mem = 1}.
      \item \texttt{specfun\_ph} asks for the frequency-dependent phonon self-energy and spectral function to be computed.
      \item \texttt{filqf} is the file containing the $\mb{q}$-points for the phonon spectral function calculation.
      \item \texttt{nkf1}, \texttt{nkf2}, and \texttt{nkf3} are the size of the (homogeneous, $\Gamma$-centered) $\mb{k}$-point grid used for the BZ integration in the self-energy.
      \item The spectral function is computed for a uniform grid of frequencies between \texttt{wmin\_specfun = 0.00} (eV) and \texttt{wmax\_specfun = 0.12} (eV) with \texttt{nw\_specfun = 200} points.
      \item \texttt{degaussw = 0.05} (eV) is the smearing parameter $\eta$ in the self-energy formula Eq.~\eqref{eq:phself}.
      \item Compared to the previous version of EPW, \texttt{temps} now only contains the low temperature $T^{\rm low}$. You may provide multiple low temperatures here.
   \end{itemize}
\end{footnotesize}


\vspace{0.5cm}

In addition to the standard output, the code will produce additional output files.
%
The frequency-dependent self-energy can be read from the file \texttt{specfun\_sup.phon}.
%
Its header reads:
{\color{cverb} \small
\begin{verbatim}
#Phonon eigenenergies + real and im part of phonon self-energy (meV)
#Q-point    Mode      Temp.[K]       smearing[eV]       w_q[eV]    w[eV]
   Real Pi(w, T_low)[meV]   Real Pi(w=0,T_high)[meV]     Im Pi(w, T_low)[meV]
\end{verbatim}
}

Each row corresponds to a $(\mb{q}, \nu, \omega)$ triplet:
\begin{footnotesize}
\begin{itemize}
   \item Column~3 (\texttt{Temp.[K]}) is the low temperature $T^{\rm low}$ specified in \texttt{temps}.
   \item Column~4 (\texttt{smearing[eV]}) is the smearing $\eta$ used in the self-energy (i.e., \texttt{degaussw} in eV).
   \item Column~5 (\texttt{w\_q[eV]}) is the bare DFPT phonon eigenenergy $\omega_\nuq$.
   \item Column~6 (\texttt{w[eV]}) is the frequency $\omega$ on the user-defined grid.
   \item Column~7 contains the real part of the dynamical low-temperature self-energy
         $\mathrm{Re}\,\Pi^{\rm 1L}_\nuq(\omega; T^{\rm low})$.
   \item Column~8 contains the real part of the static high-temperature self-energy
         $\mathrm{Re}\,\Pi^{\rm 1L}_\nuq(0; T^{\rm high})$, computed automatically with the SCF smearing.
   \item Column~9 contains the imaginary part of the dynamical low-temperature self-energy
         $\mathrm{Im}\,\Pi^{\rm 1L}_\nuq(\omega; T^{\rm low})$.
\end{itemize}
\end{footnotesize}
The unscreened self-energy entering Eq.~\eqref{eq:phspec} is then $\Delta\Pi_\nuq(\omega; T^{\rm low}) = (\mathrm{col.~7} - \mathrm{col.~8}) + i\, \mathrm{col.~9}$.
